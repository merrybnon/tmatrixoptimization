% Transition-matrix VAE (TMVAE in src/tm_ml/models.py) as a block map: the whole
% VAE along the top, the latent split into its per-node and graph-level parts,
% boxed together, and the property predictor in a row beneath the decoder.
% Both latents feed both the decoder and the predictor, so the arrows enter
% and leave the box rather than either pill, and the predictor's elbow taps
% the box's one outgoing arrow at a junction dot.
\documentclass[border=8pt, multi=tikzpicture]{standalone}
\usepackage{arch}

\begin{document}
\begin{tikzpicture}[
  big/.style  = {minimum width=2.3cm, minimum height=2.4cm, font=\small},
  pill/.style = {rounded corners=0.35cm, minimum width=1.6cm, minimum height=0.7cm,
                 font=\small, dash pattern=on 2pt off 1.5pt},
  row/.style  = {flow},
]

% ----------------------------------------------------------------------- vae --
\node[iob, big]  (x)  at ( 0.00, 0) {Transition\\Matrix $T$};
\node[enc, big]  (eg) at ( 2.75, 0) {Encoder\\Graph\\Transformer};
\node[lat, pill, rounded corners=0.4cm, minimum height=0.8cm, minimum width=2.3cm,
      dash pattern=]
                 (zn) at ( 5.80, 0.6) {node latent $z_i$};
\node[lat, pill, rounded corners=0.4cm, minimum height=0.8cm, minimum width=2.3cm,
      dash pattern=]
                 (zg) at ( 5.80, -0.6) {global latent $z_G$};
\node[enc, big]  (dt) at ( 9.45, 0) {Decoder\\Transformer};
\node[enc, big]  (hd) at (12.20, 0) {Pair Head};
\node[iob, big]  (xh) at (14.95, 0) {output $\hat{T}$};
\begin{scope}[on background layer]
  \node[blk, fill=white, draw=latLine, rounded corners=4pt, inner sep=0.2cm,
        fit=(zn) (zg)] (zb) {};
\end{scope}

% ----------------------------------------------------------------- predictor --
\node[enc, big]  (pr) at ( 9.45, -3.6) {Predictor};
\node[iob, big]  (yh) at (12.20, -3.6) {output $\log\hat{y}$};

% -------------------------------------------------------------------- arrows --
\draw[row] (x)  -- (eg);
\draw[row] (eg) -- (zb);
\draw[row] (zb) -- (dt);
\draw[row] (dt) -- (hd);
\draw[row] (hd) -- (xh);
\draw[row] (7.75, 0) -- (7.75, -3.6) -- (pr.west);
\fill[black!62] (7.75, 0) circle (1.6pt);
\draw[row] (pr) -- (yh);

\end{tikzpicture}
\end{document}
