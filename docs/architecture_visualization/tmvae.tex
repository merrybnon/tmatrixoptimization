% Transition-matrix VAE (TMVAE in src/tm_ml/models.py) as a block map: the whole
% VAE along the top, the latent split into its per-node and graph-level parts,
% and the property predictor in a row beneath the decoder. Both latents feed
% both the decoder and the predictor, so the predictor's elbow taps the two
% decoder arrows at junction dots rather than leaving either pill.
\documentclass[border=8pt, multi=tikzpicture]{standalone}
\usepackage{arch}

\begin{document}
\begin{tikzpicture}[
  big/.style  = {minimum width=2.3cm, minimum height=2.4cm, font=\small},
  pill/.style = {rounded corners=0.35cm, minimum width=1.6cm, minimum height=0.7cm,
                 font=\small, dash pattern=on 2pt off 1.5pt},
  row/.style  = {flow},
]

% ----------------------------------------------------------------------- vae --
\node[iob, big]  (x)  at ( 0.00, 0) {Transition\\Matrix $T$};
\node[enc, big]  (eg) at ( 2.75, 0) {Encoder\\Graph\\Transformer};
\node[lat, pill, rounded corners=0.4cm, minimum height=0.8cm, minimum width=2.3cm,
      dash pattern=]
                 (zn) at ( 5.50, 0.6) {node latent $z_i$};
\node[lat, pill, rounded corners=0.4cm, minimum height=0.8cm, minimum width=2.3cm,
      dash pattern=]
                 (zg) at ( 5.50, -0.6) {global latent $z_G$};
\node[enc, big]  (dt) at ( 9.05, 0) {Decoder\\Transformer};
\node[enc, big]  (hd) at (11.80, 0) {Pair Head};
\node[iob, big]  (xh) at (14.55, 0) {output $\hat{T}$};

% ----------------------------------------------------------------- predictor --
\node[enc, big]  (pr) at ( 9.05, -3.6) {Predictor};
\node[iob, big]  (yh) at (11.80, -3.6) {output $\log\hat{y}$};

% -------------------------------------------------------------------- arrows --
\draw[row] (x)  -- (eg);
\draw[row] (eg.east |- zn) -- (zn);
\draw[row] (eg.east |- zg) -- (zg);
\draw[row] (zn) -- (dt.west |- zn);
\draw[row] (zg) -- (dt.west |- zg);
\draw[row] (dt) -- (hd);
\draw[row] (hd) -- (xh);
\draw[row] (7.05, 0.6) -- (7.05, -3.6) -- (pr.west);
\fill[black!62] (7.05, 0.6) circle (1.6pt) (7.05, -0.6) circle (1.6pt);
\draw[row] (pr) -- (yh);

\end{tikzpicture}
\end{document}
